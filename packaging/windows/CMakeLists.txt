cmake_minimum_required(VERSION 3.22)
project(RelayWindowsSmoke LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Run the native ConPTY smoke test on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core)
add_executable(relay-pty-smoke pty-smoke.cpp ../../engine/pty/PtyWin.cpp)
target_link_libraries(relay-pty-smoke PRIVATE Qt6::Core)
target_compile_definitions(relay-pty-smoke PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
if(MSVC)
  target_compile_options(relay-pty-smoke PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_test(NAME native-conpty COMMAND relay-pty-smoke)
set_tests_properties(native-conpty PROPERTIES TIMEOUT 60)
